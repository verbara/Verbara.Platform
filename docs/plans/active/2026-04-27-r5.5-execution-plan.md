# R5.5 P0L.3 — observability side-stack scrapping docker-compose.full.yml services.

networks:
  default:
    external: true
    name: asterisk-platform_default

services:
  prometheus:
    image: prom/prometheus:v3.2.1
    container_name: r55-prometheus
    volumes:
      - ./observability/prometheus.yml:/etc/prometheus/prometheus.yml:ro
      - ../docs/operations/alerts.yml:/etc/prometheus/alerts.yml:ro
      - prometheus-data:/prometheus
    command:
      - --config.file=/etc/prometheus/prometheus.yml
      - --storage.tsdb.path=/prometheus
      - --storage.tsdb.retention.time=30d
    ports:
      - "9090:9090"
    healthcheck:
      test: ["CMD", "wget", "-qO-", "http://localhost:9090/-/healthy"]
      interval: 10s

  grafana:
    image: grafana/grafana:11.5.2
    container_name: r55-grafana

      GF_SECURITY_ADMIN_PASSWORD: r55-staging

      - ./observability/grafana-provisioning:/etc/grafana/provisioning:ro
      - grafana-data:/var/lib/grafana
    ports:
      - "3000:3000"
    depends_on:
      - prometheus
      - loki

  loki:
    image: grafana/loki:3.3.2
    container_name: r55-loki
    volumes:
      - ./observability/loki-config.yml:/etc/loki/local-config.yaml:ro
      - loki-data:/loki
    command: -config.file=/etc/loki/local-config.yaml

      - "3100:3100"

    #!substdef "!MY_SIP_PORT!{{ .Values.kamailio.sipPort }}!g"

    alias=MY_HOSTNAME

    loadmodule "rr.so"
    loadmodule "sl.so"
    loadmodule "dispatcher.so"
    loadmodule "rtpengine.so"
    loadmodule "siputils.so"

    modparam("dispatcher", "list_file", "/etc/kamailio/dispatcher.list")
    modparam("dispatcher", "ds_ping_interval", 30)
    modparam("dispatcher", "ds_probing_mode", 1)
    modparam("dispatcher", "flags", 2)

    request_route {
      if (!mf_process_maxfwd_header("10")) {
        sl_send_reply("483","Too Many Hops"); exit;
      }
      if (is_method("REGISTER")) {
        sl_send_reply("404","Registrations Forwarded — see SIPp scenarios"); exit;
      }
      if (has_totag()) { route(WITHIN_DLG); exit; }
      record_route();
      if (is_method("INVITE")) {
        if (!ds_select_dst("1", "0")) {  # Sticky session by call-id
          sl_send_reply("503","No backend Asterisk available"); exit;
        }
        rtpengine_manage();
        t_on_reply("MANAGE_REPLY");
      }
      if (!t_relay()) { sl_reply_error(); exit; }

    onreply_route[MANAGE_REPLY] {
      if (status =~ "(180)|(183)|(2[0-9][0-9])") { rtpengine_manage(); }
    }

    route[WITHIN_DLG] {
      if (loose_route()) {
        if (is_method("BYE")) { rtpengine_delete(); }
        else if (is_method("INVITE")) { rtpengine_manage(); }
        if (!t_relay()) { sl_reply_error(); exit; }

#!/usr/bin/env bash
# scripts/seed-staging.sh — R5.5 multi-tenant staging seed
# Targets PLATFORM_API_URL with admin token. Creates 3 tenants with realistic load.
set -euo pipefail

PLATFORM_API_URL="${PLATFORM_API_URL:-http://localhost:5000}"
ADMIN_TOKEN="${ADMIN_TOKEN:-}"
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

if [ -z "$ADMIN_TOKEN" ]; then
    echo "[seed] Bootstrapping setup wizard to obtain admin token..."
    SETUP_RESPONSE=$(curl -fsS -X POST "$PLATFORM_API_URL/api/v1/setup" \
        -H "Content-Type: application/json" \
        -d '{"adminEmail":"platform@admin.local","adminPassword":"PlatformAdmin2026!","tenantName":"r55-staging"}')
    ADMIN_TOKEN=$(echo "$SETUP_RESPONSE" | jq -r '.managementApiKey // .accessToken')
fi

if [ -z "$ADMIN_TOKEN" ] || [ "$ADMIN_TOKEN" = "null" ]; then
    echo "[seed] FAIL: could not obtain admin token. Stack may not be running or already configured."
    exit 1
fi

for profile in small medium large; do
    echo "[seed] Creating $profile-loadtest tenant..."
    PROFILE_FILE="$SCRIPT_DIR/seed-data/$profile-tenant.json"
    TENANT_ID=$(jq -r '.tenantId' "$PROFILE_FILE")
    AGENTS=$(jq -r '.agents' "$PROFILE_FILE")
    QUEUES=$(jq -r '.queues' "$PROFILE_FILE")
    NAME=$(jq -r '.name' "$PROFILE_FILE")

    # Create tenant
    curl -fsS -X POST "$PLATFORM_API_URL/api/v1/management/tenants" \
        -H "Authorization: Bearer $ADMIN_TOKEN" \
        -H "Content-Type: application/json" \
        -d "{\"tenantId\":\"$TENANT_ID\",\"name\":\"$NAME\"}" || echo "  (already exists, skipping)"

    # Create agents
    for i in $(seq 1 $AGENTS); do
        curl -fsS -X POST "$PLATFORM_API_URL/api/v1/admin/agents" \
            -H "Authorization: Bearer $ADMIN_TOKEN" \
            -H "X-Tenant-Id: $TENANT_ID" \
            -H "Content-Type: application/json" \
            -d "{\"name\":\"Agent $i\",\"email\":\"agent$i@$TENANT_ID.local\"}" >/dev/null || true
    done

    # Create queues
    for i in $(seq 1 $QUEUES); do
        curl -fsS -X POST "$PLATFORM_API_URL/api/v1/queues" \
            -H "Authorization: Bearer $ADMIN_TOKEN" \
            -H "X-Tenant-Id: $TENANT_ID" \
            -H "Content-Type: application/json" \
            -d "{\"name\":\"queue-$i\",\"strategy\":\"ringall\",\"maxWait\":60}" >/dev/null || true
    done

    echo "[seed]   ✓ $TENANT_ID seeded ($AGENTS agents, $QUEUES queues)"
done

echo "[seed] Done. Use ADMIN_TOKEN=$ADMIN_TOKEN for subsequent ops."

# scripts/sipp-test.sh — Run SIPp suite for R5.5

REPORT_DIR="$ROOT/sipp-reports/$(date +%Y%m%d-%H%M%S)"

for scenario in 01-basic-call 02-ivr-navigation 03-queue-join 04-conference 05-transfer; do

    sipp -sf "$SCENARIO_DIR/$scenario.xml" \
         -s queue-1 \
         -d 5000 -l 10 -r 1 -m 100 \

         -trace_screen -screen_file "$REPORT_DIR/$scenario.log" \
         "$TARGET:5060" || echo "[sipp]   ⚠ $scenario had failures (check logs)"
done

echo "[sipp] Reports: $REPORT_DIR"

- [ ] **Step 4: Create scripts/chaos-test.sh**

```bash
#!/usr/bin/env bash
# scripts/chaos-test.sh — Run Pumba chaos suite with snapshots
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
ROOT="$(dirname "$SCRIPT_DIR")"
EXP_DIR="$ROOT/tests/chaos-experiments/pumba"
SNAP_DIR="$ROOT/chaos-snapshots/$(date +%Y%m%d-%H%M%S)"
mkdir -p "$SNAP_DIR"

snapshot() {
    local name=$1
    docker compose -f "$ROOT/docker/docker-compose.full.yml" exec -T postgres pg_dumpall -U postgres > "$SNAP_DIR/$name-pg.sql" 2>/dev/null || true
    docker compose -f "$ROOT/docker/docker-compose.full.yml" exec -T redis redis-cli SAVE >/dev/null 2>&1 || true
    docker cp "$(docker ps -qf name=redis)":/data/dump.rdb "$SNAP_DIR/$name-redis.rdb" 2>/dev/null || true
}

for exp in "$EXP_DIR"/*.sh; do
    NAME=$(basename "$exp" .sh)
    echo "[chaos] === $NAME ==="
    snapshot "${NAME}-pre"
    "$exp" || echo "[chaos]   ⚠ $NAME ended non-zero (expected for some)"
    sleep 30  # Recovery window
    snapshot "${NAME}-post"
done

echo "[chaos] Snapshots: $SNAP_DIR"
```
